3 Step 1: Scale and Run PCA

# Extract the numeric variables (drop the cultivar label for PCA)
X <- wine_clean |>
  select(where(is.numeric)) |>
  as.matrix()

# Standardize
X_s <- scale(X)

# Run PCA
pca_wine <- prcomp(X_s, center = FALSE, scale. = FALSE)

# Proportion of variance explained by each PC
